    68→    68→function formatTimeAgo(dateStr: string): string {
    69→    69→  const diff = Date.now() - new Date(dateStr).getTime();
    70→    70→  const mins = Math.floor(diff / 60000);
    71→    71→  if (mins < 1) return "just now";
    72→    72→  if (mins < 60) return `${mins}m ago`;
    73→    73→  const hrs = Math.floor(mins / 60);
    74→    74→  if (hrs < 24) return `${hrs}h ago`;
    75→    75→  const days = Math.floor(hrs / 24);
    76→    76→  return `${days}d ago`;
    77→    77→}

    79→    79→function getAvatarColor(name: string): string {
    80→    80→  const colors = [
    81→    81→    "from-[#1A237E] to-[#283593]",
    82→    82→    "from-[#D32F2F] to-[#E53935]",
    83→    83→    "from-[#2E7D32] to-[#388E3C]",
    84→    84→    "from-[#E65100] to-[#F57C00]",
    85→    85→    "from-[#6A1B9A] to-[#8E24AA]",
    86→    86→    "from-[#00695C] to-[#00897B]",
    87→    87→  ];
    88→    88→  let hash = 0;
    89→    89→  for (let i = 0; i < name.length; i++) {
    90→    90→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    91→    91→  }
    92→    92→  return colors[Math.abs(hash) % colors.length];
    93→    93→}
