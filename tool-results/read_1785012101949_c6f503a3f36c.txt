    64→/* ============ HELPERS ============ */
    65→function formatTimeAgo(dateStr: string): string {
    66→  const diff = Date.now() - new Date(dateStr).getTime();
    67→  const mins = Math.floor(diff / 60000);
    68→  if (mins < 1) return "just now";
    69→  if (mins < 60) return `${mins}m ago`;
    70→  const hrs = Math.floor(mins / 60);
    71→  if (hrs < 24) return `${hrs}h ago`;
    72→  const days = Math.floor(hrs / 24);
    73→  return `${days}d ago`;
    74→}

    76→function getAvatarColor(name: string): string {
    77→  const colors = [
    78→    "from-[#1A237E] to-[#283593]",
    79→    "from-[#D32F2F] to-[#E53935]",
    80→    "from-[#2E7D32] to-[#388E3C]",
    81→    "from-[#E65100] to-[#F57C00]",
    82→    "from-[#6A1B9A] to-[#8E24AA]",
    83→    "from-[#00695C] to-[#00897B]",
    84→  ];
    85→  let hash = 0;
    86→  for (let i = 0; i < name.length; i++) {
    87→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    88→  }
    89→  return colors[Math.abs(hash) % colors.length];
    90→}

   558→                            <p className="text-[13px] text-gray-700 leading-relaxed break-words pr-16">
   559→                              {msg.content}
   560→                            </p>
   561→                          </div>
   562→
   563→                          {/* Reactions */}
   564→                          <MessageReactions
   565→                            reactions={msg.reactions}
   566→                            onReact={(r) => addReaction(msg.id, r)}
   567→                            showPicker={showReactions === msg.id}
   568→                            onTogglePicker={() =>
   569→                              setShowReactions(showReactions === msg.id ? null : msg.id)
   570→                            }
   571→                          />
   572→
   573→                          {/* Hover actions */}
   574→                          <div className="flex items-center gap-0.5 mt-0.5 opacity-0 group-hover:opacity-100 transition-opacity">
   575→                            <button
   576→                              onClick={() => setReplyingTo(msg)}
   577→                              className="p-1 hover:bg-gray-100 rounded-md text-gray-400 hover:text-[#1A237E] transition-colors"
   578→                              title="Reply"
   579→                            >
   580→                              <Reply className="size-3" />
   581→                            </button>
   582→                            <button
   583→                              onClick={() => setShowReactions(showReactions === msg.id ? null : msg.id)}
   584→                              className="p-1 hover:bg-gray-100 rounded-md text-gray-400 hover:text-[#1A237E] transition-colors"
   585→                              title="React"
   586→                            >
   587→                              <SmilePlus className="size-3" />
   588→                            </button>
   589→                            <button
   590→                              onClick={() => setReportingMsg(msg)}
   591→                              className="p-1 hover:bg-red-50 rounded-md text-gray-400 hover:text-red-500 transition-colors"
   592→                              title="Report"
   593→                            >
   594→                              <Flag className="size-3" />
   595→                            </button>
   596→                          </div>
   597→                        </div>
   598→                      </div>
   599→                    )}
   600→                  </div>
   601→                );
   602→              })}
   603→            </div>
   604→            <div ref={messagesEndRef} />
   605→          </div>
   606→
   607→          {/* Error banner */}
   608→          <AnimatePresence>
   609→            {error && (
   610→              <motion.div
   611→                initial={{ opacity: 0, height: 0 }}
   612→                animate={{ opacity: 1, height: "auto" }}
   613→                exit={{ opacity: 0, height: 0 }}
   614→                className="bg-red-50 border-t border-red-100 px-4 py-2 flex items-center gap-2 shrink-0"
   615→              >
   616→                <AlertTriangle className="size-3.5 text-red-500 shrink-0" />
   617→                <p className="text-xs text-red-600 flex-1">{error}</p>
   618→                <button
   619→                  onClick={() => setError("")}
   620→                  className="text-red-400 hover:text-red-600"
   621→                >
   622→                  <X className="size-3" />
   623→                </button>
   624→              </motion.div>
   625→            )}
   626→          </AnimatePresence>
   627→
   628→          {/* Input Area */}
   629→          {activeChannel && (
   630→            <div className="bg-white border-t border-gray-100 p-3 shrink-0">
   631→              {/* Replying to indicator */}
   632→              <AnimatePresence>
   633→                {replyingTo && (
   634→                  <motion.div
   635→                    initial={{ opacity: 0, height: 0 }}
   636→                    animate={{ opacity: 1, height: "auto" }}
   637→                    exit={{ opacity: 0, height: 0 }}
   638→                    className="overflow-hidden"
   639→                  >
   640→                    <div className="flex items-center gap-2 bg-[#F0F4FF] border border-[#1A237E]/10 rounded-lg px-3 py-2 mb-2">
   641→                      <Reply className="size-3 text-[#1A237E] shrink-0" />
   642→                      <div className="min-w-0 flex-1">
   643→                        <p className="text-[10px] font-semibold text-[#1A237E]/60">
   644→                          Replying to {replyingTo.displayName}
   645→                        </p>
   646→                        <p className="text-[10px] text-gray-500 truncate">
   647→                          {replyingTo.content.substring(0, 80)}
   648→                        </p>
   649→                      </div>
   650→                      <button
   651→                        onClick={() => setReplyingTo(null)}
   652→                        className="text-gray-400 hover:text-gray-600 shrink-0"
   653→                      >
   654→                        <X className="size-3.5" />
   655→                      </button>
   656→                    </div>
   657→                  </motion.div>
   658→                )}
   659→              </AnimatePresence>
   660→
   661→              {/* Composer */}
   662→              <div className="flex items-end gap-2">
   663→                <div className="flex-1 relative">
   664→                  <textarea
   665→                    ref={inputRef}
   666→                    value={inputText}
   667→                    onChange={handleInputChange}
   668→                    onKeyDown={handleKeyDown}
   669→                    placeholder={
   670→                      activeChannel.isAnnouncement
   671→                        ? "This channel is for announcements only."
   672→                        : `Message ${activeChannel.name}...`
   673→                    }
   674→                    disabled={activeChannel.isAnnouncement}
   675→                    rows={1}
   676→                    className="w-full resize-none rounded-xl border border-gray-200 bg-[#FAFBFF] px-3.5 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-[#1A237E]/15 focus:border-[#1A237E]/25 transition-all min-h-[42px] max-h-24 disabled:opacity-50 disabled:cursor-not-allowed"
   677→                    style={{ lineHeight: "1.5" }}
   678→                  />
   679→                  {/* Character counter */}
   680→                  {charCount > 0 && (
   681→                    <span
   682→                      className={`absolute bottom-1.5 right-3 text-[9px] ${
   683→                        charCount > MAX_CHARS * 0.9
   684→                          ? "text-red-400"
   685→                          : "text-gray-300"
   686→                      }`}
   687→                    >
   688→                      {charCount}/{MAX_CHARS}
   689→                    </span>
   690→                  )}
   691→                </div>
   692→                <button
   693→                  onClick={sendMessage}
   694→                  disabled={!inputText.trim() || sending || activeChannel.isAnnouncement}
   695→                  className="shrink-0 w-10 h-[42px] rounded-xl bg-[#1A237E] hover:bg-[#283593] text-white flex items-center justify-center transition-colors disabled:opacity-30 disabled:cursor-not-allowed shadow-sm"
   696→                >
   697→                  {sending ? (
   698→                    <Loader2 className="size-4 animate-spin" />
   699→                  ) : (
   700→                    <Send className="size-4" />
   701→                  )}
   702→                </button>
   703→              </div>
   704→            </div>
   705→          )}
   706→        </div>
   707→      </div>
   708→
   709→      {/* ===== REPORT MODAL ===== */}
   710→      <AnimatePresence>
   711→        {reportingMsg && (
   712→          <motion.div
   713→            initial={{ opacity: 0 }}
   714→            animate={{ opacity: 1 }}
   715→            exit={{ opacity: 0 }}
   716→            className="fixed inset-0 bg-black/40 z-50 flex items-center justify-center p-4"
   717→            onClick={() => {
   718→              setReportingMsg(null);
   719→              setReportReason("");
   720→              setReportExplanation("");
   721→            }}
   722→          >
   723→            <motion.div
   724→              initial={{ scale: 0.95, opacity: 0, y: 10 }}
   725→              animate={{ scale: 1, opacity: 1, y: 0 }}
   726→              exit={{ scale: 0.95, opacity: 0, y: 10 }}
   727→              className="bg-white rounded-2xl p-5 max-w-sm w-full shadow-xl"
   728→              onClick={(e) => e.stopPropagation()}
   729→            >
   730→              {/* Header */}
   731→              <div className="flex items-center gap-3 mb-4">
   732→                <div className="w-10 h-10 rounded-full bg-red-50 flex items-center justify-center">
   733→                  <Flag className="size-5 text-red-500" />
   734→                </div>
   735→                <div>
   736→                  <h3 className="text-sm font-bold text-gray-800">
   737→                    Report Message
   738→                  </h3>
   739→                  <p className="text-[10px] text-gray-500">
   740→                    Reviewed confidentially by a church moderator
   741→                  </p>
   742→                </div>
   743→              </div>
   744→
   745→              {/* Reported message preview */}
   746→              <div className="bg-gray-50 rounded-lg p-2.5 mb-4 border border-gray-100">
   747→                <p className="text-[11px] text-gray-600 italic line-clamp-2">
   748→                  &ldquo;{reportingMsg.content.substring(0, 120)}&rdquo;
   749→                </p>
   750→              </div>
   751→
   752→              {/* Reason selection */}
   753→              <div className="mb-4">
   754→                <p className="text-xs font-medium text-gray-700 mb-2">
   755→                  What is the reason?
   756→                </p>
   757→                <div className="space-y-1.5 max-h-48 overflow-y-auto">
   758→                  {REPORT_REASONS.map((r) => (
   759→                    <label
   760→                      key={r.value}
   761→                      className={`flex items-center gap-2.5 px-3 py-2 rounded-lg border cursor-pointer transition-colors ${
   762→                        reportReason === r.value
   763→                          ? "border-[#1A237E] bg-[#1A237E]/5"
   764→                          : "border-gray-100 hover:bg-gray-50"
   765→                      }`}
   766→                    >
   767→                      <div
   768→                        className={`w-4 h-4 rounded-full border-2 flex items-center justify-center transition-colors ${
   769→                          reportReason === r.value
   770→                            ? "border-[#1A237E]"
   771→                            : "border-gray-300"
   772→                        }`}
   773→                      >
   774→                        {reportReason === r.value && (
   775→                          <div className="w-2 h-2 rounded-full bg-[#1A237E]" />
   776→                        )}
   777→                      </div>
   778→                      <span className="text-xs text-gray-700">{r.label}</span>
   779→                    </label>
   780→                  ))}
   781→                </div>
   782→              </div>
   783→
   784→              {/* Optional explanation */}
   785→              {reportReason && (
   786→                <div className="mb-4">
   787→                  <p className="text-xs font-medium text-gray-700 mb-1.5">
   788→                    Additional details (optional)
   789→                  </p>
   790→                  <textarea
   791→                    value={reportExplanation}
   792→                    onChange={(e) => setReportExplanation(e.target.value)}
   793→                    placeholder="Provide any helpful context..."
   794→                    rows={2}
   795→                    className="w-full resize-none rounded-lg border border-gray-200 bg-gray-50 px-3 py-2 text-xs focus:outline-none focus:ring-2 focus:ring-[#1A237E]/15 focus:border-[#1A237E]/25 transition-colors"
   796→                  />
   797→                </div>
   798→              )}
   799→
   800→              {/* Actions */}
   801→              <div className="flex gap-3">
   802→                <button
   803→                  onClick={() => {
   804→                    setReportingMsg(null);
   805→                    setReportReason("");
   806→                    setReportExplanation("");
   807→                  }}
   808→                  className="flex-1 py-2.5 rounded-xl border border-gray-200 text-xs font-medium text-gray-600 hover:bg-gray-50 transition-colors"
   809→                >
   810→                  Cancel
   811→                </button>
   812→                <button
   813→                  onClick={submitReport}
   814→                  disabled={!reportReason || reportSubmitting}
   815→                  className="flex-1 py-2.5 rounded-xl bg-red-500 hover:bg-red-600 text-xs font-medium text-white transition-colors disabled:opacity-40 disabled:cursor-not-allowed flex items-center justify-center gap-1.5"
   816→                >
   817→                  {reportSubmitting ? (
   818→                    <Loader2 className="size-3 animate-spin" />
   819→                  ) : (
   820→                    <Flag className="size-3" />
   821→                  )}
   822→                  Submit Report
   823→                </button>
   824→              </div>
   825→            </motion.div>
   826→          </motion.div>
   827→        )}
   828→      </AnimatePresence>
   829→    </div>
   830→  );
   831→}
