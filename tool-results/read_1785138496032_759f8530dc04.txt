   270→   270→  const item = moderationQueue.find((m) => m.id === id);
   271→   271→  if (!item) return false;
   272→   272→  item.action = "approved";
   273→   273→

   275→   275→  switch (targetCollection) {
   276→   276→    case "publicPrayers": {
   277→   277→      const prayer = publicPrayers.find((p) => p.id === item.targetId);
   278→   278→      if (prayer) prayer.status = "approved";
