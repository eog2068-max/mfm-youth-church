    25→export function getPresenceCount(): number {
    26→  const now = Date.now();
    27→  const threshold = 5 * 60 * 1000; // 5 minutes
    28→  let count = 0;
    29→  presenceStore.forEach((ts) => {
    30→    if (now - ts < threshold) count++;
    31→  });
    32→  // Add a base count to simulate real activity
    33→  return count + Math.floor(Math.random() * 15) + 20;
    34→}

    36→export function recordPresence(sessionId: string): void {
    37→  presenceStore.set(sessionId, Date.now());
    38→  // Clean up old entries
    39→  const now = Date.now();
    40→  const threshold = 10 * 60 * 1000;
    41→  presenceStore.forEach((ts, key) => {
    42→    if (now - ts > threshold) presenceStore.delete(key);
    43→  });
    44→}

   268→export function approveModerationItem(id: string, targetCollection: "publicPrayers" | "amenPosts" | "questionResponses"): boolean {
   269→  // Find in moderation queue and update status
   270→  const item = moderationQueue.find((m) => m.id === id);
   271→  if (!item) return false;
   272→  item.action = "approved";
   273→
   274→  // Find and approve in the target collection
   275→  switch (targetCollection) {
   276→    case "publicPrayers": {
   277→      const prayer = publicPrayers.find((p) => p.id === item.targetId);
   278→      if (prayer) prayer.status = "approved";
   279→      break;
   280→    }
   281→    case "amenPosts": {
   282→      const post = amenPosts.find((p) => p.id === item.targetId);
   283→      if (post) post.status = "approved";
   284→      break;
   285→    }
   286→    case "questionResponses": {
   287→      const resp = questionResponses.find((r) => r.id === item.targetId);
   288→      if (resp) resp.status = "approved";
   289→      break;
   290→    }
   291→  }
   292→  return true;
   293→}

   295→export function hideModerationItem(id: string, targetCollection: "publicPrayers" | "amenPosts" | "questionResponses"): boolean {
   296→  const item = moderationQueue.find((m) => m.id === id);
   297→  if (!item) return false;
   298→  item.action = "hidden";
   299→
   300→  switch (targetCollection) {
   301→    case "publicPrayers": {
   302→      const prayer = publicPrayers.find((p) => p.id === item.targetId);
   303→      if (prayer) prayer.status = "hidden";
   304→      break;
   305→    }
   306→    case "amenPosts": {
   307→      const post = amenPosts.find((p) => p.id === item.targetId);
   308→      if (post) post.status = "hidden";
   309→      break;
   310→    }
   311→    case "questionResponses": {
   312→      const resp = questionResponses.find((r) => r.id === item.targetId);
   313→      if (resp) resp.status = "hidden";
   314→      break;
   315→    }
   316→  }
   317→  return true;
   318→}

   741→export function getChatMessages(
   742→  channelId: string,
   743→  limit: number = 50,
   744→  before?: string
   745→): ChatMessage[] {
   746→  let messages = chatMessages.filter(
   747→    (m) => m.channelId === channelId && m.status === "approved"
   748→  );
   749→  if (before) {
   750→    const beforeDate = new Date(before).getTime();
   751→    messages = messages.filter((m) => new Date(m.createdAt).getTime() < beforeDate);
   752→  }
   753→  return messages.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()).slice(0, limit);
   754→}

   756→export function addChatMessage(
   757→  channelId: string,
   758→  content: string,
   759→  sessionId: string,
   760→  displayName: string = "Anonymous",
   761→  replyToId?: string
   762→): ChatMessage {
   763→  const channel = chatChannels.find((ch) => ch.id === channelId);
   764→  const isAnnouncement = channel?.isAnnouncement || false;
   765→
   766→  const msg: ChatMessage = {
   767→    id: `msg_${Date.now()}_${Math.random().toString(36).substring(2, 7)}`,
   768→    channelId,
   769→    content: content.substring(0, 500),
   770→    displayName,
   771→    sessionId,
   772→    status: "approved",
   773→    isAnnouncement,
   774→    replyToId,
   775→    reactions: {},
   776→    reportCount: 0,
   777→    createdAt: new Date().toISOString(),
   778→  };
   779→  chatMessages.push(msg);
   780→
   781→  // Update channel stats
   782→  if (channel) {
   783→    channel.lastMessageAt = msg.createdAt;
   784→    channel.messageCount++;
   785→    channel.updatedAt = msg.createdAt;
   786→  }
   787→
   788→  return msg;
   789→}

   791→export function addReactionToChatMessage(
   792→  messageId: string,
   793→  reaction: string
   794→): ChatMessage | null {
   795→  const msg = chatMessages.find((m) => m.id === messageId);
   796→  if (msg) {
   797→    msg.reactions[reaction] = (msg.reactions[reaction] || 0) + 1;
   798→    return msg;
   799→  }
   800→  return null;
   801→}

   803→export function reportChatMessage(
   804→  messageId: string,
   805→  channelId: string,
   806→  sessionId: string,
   807→  reason: string
   808→): ChatReport {
   809→  const report: ChatReport = {
   810→    id: `rpt_${Date.now()}`,
   811→    messageId,
   812→    channelId,
   813→    sessionId,
   814→    reason,
   815→    status: "pending",
   816→    createdAt: new Date().toISOString(),
   817→  };
   818→  chatReports.push(report);
   819→
   820→  // Increment report count on message
   821→  const msg = chatMessages.find((m) => m.id === messageId);
   822→  if (msg) msg.reportCount++;
   823→
   824→  // Auto-hide if 3+ reports
   825→  if (msg && msg.reportCount >= 3) {
   826→    msg.status = "hidden";
   827→  }
   828→
   829→  return report;
   830→}

   838→// Admin functions for FamilyChat
   839→export function removeChatMessage(messageId: string): boolean {
   840→  const idx = chatMessages.findIndex((m) => m.id === messageId);
   841→  if (idx !== -1) {
   842→    chatMessages[idx].status = "hidden";
   843→    return true;
   844→  }
   845→  return false;
   846→}

   848→export function pinChatMessage(messageId: string): boolean {
   849→  const msg = chatMessages.find((m) => m.id === messageId);
   850→  if (msg) {
   851→    msg.status = "pinned";
   852→    msg.pinnedAt = new Date().toISOString();
   853→    return true;
   854→  }
   855→  return false;
   856→}

   858→export function unpinChatMessage(messageId: string): boolean {
   859→  const msg = chatMessages.find((m) => m.id === messageId);
   860→  if (msg && msg.status === "pinned") {
   861→    msg.status = "approved";
   862→    msg.pinnedAt = undefined;
   863→    return true;
   864→  }
   865→  return false;
   866→}
