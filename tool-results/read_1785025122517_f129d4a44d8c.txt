    68→function formatTimeAgo(dateStr: string): string {
    69→  const diff = Date.now() - new Date(dateStr).getTime();
    70→  const mins = Math.floor(diff / 60000);
    71→  if (mins < 1) return "just now";
    72→  if (mins < 60) return `${mins}m ago`;
    73→  const hrs = Math.floor(mins / 60);
    74→  if (hrs < 24) return `${hrs}h ago`;
    75→  const days = Math.floor(hrs / 24);
    76→  return `${days}d ago`;
    77→}

    79→function getAvatarColor(name: string): string {
    80→  const colors = [
    81→    "from-[#1A237E] to-[#283593]",
    82→    "from-[#D32F2F] to-[#E53935]",
    83→    "from-[#2E7D32] to-[#388E3C]",
    84→    "from-[#E65100] to-[#F57C00]",
    85→    "from-[#6A1B9A] to-[#8E24AA]",
    86→    "from-[#00695C] to-[#00897B]",
    87→  ];
    88→  let hash = 0;
    89→  for (let i = 0; i < name.length; i++) {
    90→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    91→  }
    92→  return colors[Math.abs(hash) % colors.length];
    93→}

   561→                      /* === REGULAR MESSAGE === */
   562→                      <div className="flex gap-2.5">
   563→                        {/* Avatar */}
   564→                        <div className="w-8 h-8 rounded-full bg-gradient-to-br {getAvatarColor(msg.displayName)} flex items-center justify-center shrink-0 mt-0.5">
   565→                          <span className="text-white text-[11px] font-bold">
   566→                            {msg.displayName.charAt(0).toUpperCase()}
   567→                          </span>
   568→                        </div>
   569→
   570→                        <div className="min-w-0 flex-1">
   571→                          {/* Name + time (only show on first message in group) */}
   572→                          {showAvatar && (
   573→                            <div className="flex items-baseline gap-2 mb-0.5">
   574→                              <span className="text-[11px] font-semibold text-gray-700">
   575→                                {msg.displayName}
   576→                              </span>
   577→                              <span className="text-[10px] text-gray-400">
   578→                                {formatTimeAgo(msg.createdAt)}
   579→                              </span>
   580→                            </div>
   581→                          )}
   582→
   583→                          {/* Reply reference */}
   584→                          {msg.replyToId && (
   585→                            <div className="mt-1 bg-[#F0F4FF] border-l-2 border-[#1A237E]/25 rounded-r-lg px-2.5 py-1.5">
   586→                              {msg.replyToDisplayName && (
   587→                                <p className="text-[10px] font-semibold text-[#1A237E]/60">
   588→                                  {msg.replyToDisplayName}
   589→                                </p>
   590→                              )}
   591→                              <p className="text-[10px] text-gray-500 italic truncate max-w-xs">
   592→                                {msg.replyToContent || "Original message"}
   593→                              </p>
   594→                            </div>
   595→                          )}
   596→
   597→                          {/* Message bubble */}
   598→                          <div className="mt-0.5 group-hover:bg-gray-50/80 rounded-xl px-0.5 py-0.5 -mx-0.5 transition-colors">
   599→                            <p className="text-[13px] text-gray-700 leading-relaxed break-words pr-16">
   600→                              {msg.content}
   601→                            </p>
   602→                          </div>
   603→
   604→                          {/* Reactions */}
   605→                          <MessageReactions
   606→                            reactions={msg.reactions}
   607→                            onReact={(r) => addReaction(msg.id, r)}
   608→                            showPicker={showReactions === msg.id}
   609→                            onTogglePicker={() =>
   610→                              setShowReactions(showReactions === msg.id ? null : msg.id)
   611→                            }
   612→                          />
   613→
   614→                          {/* Hover actions */}
   615→                          <div className="flex items-center gap-0.5 mt-0.5 opacity-0 group-hover:opacity-100 transition-opacity">
   616→                            <button
   617→                              onClick={() => setReplyingTo(msg)}
   618→                              className="p-1 hover:bg-gray-100 rounded-md text-gray-400 hover:text-[#1A237E] transition-colors"
   619→                              title="Reply"
   620→                            >
   621→                              <Reply className="size-3" />
   622→                            </button>
   623→                            <button
   624→                              onClick={() => setShowReactions(showReactions === msg.id ? null : msg.id)}
   625→                              className="p-1 hover:bg-gray-100 rounded-md text-gray-400 hover:text-[#1A237E] transition-colors"
   626→                              title="React"
   627→                            >
   628→                              <SmilePlus className="size-3" />
   629→                            </button>
   630→                            <button
   631→                              onClick={() => setReportingMsg(msg)}
   632→                              className="p-1 hover:bg-red-50 rounded-md text-gray-400 hover:text-red-500 transition-colors"
   633→                              title="Report"
   634→                            >
   635→                              <Flag className="size-3" />
   636→                            </button>
   637→                          </div>
   638→                        </div>
   639→                      </div>
   640→                    )}
   641→                  </div>
   642→                );
   643→              })}
   644→            </div>
   645→            <div ref={messagesEndRef} />
   646→          </div>
   647→
   648→          {/* Error banner */}
   649→          <AnimatePresence>
   650→            {error && (
   651→              <motion.div
   652→                initial={{ opacity: 0, height: 0 }}
   653→                animate={{ opacity: 1, height: "auto" }}
   654→                exit={{ opacity: 0, height: 0 }}
   655→                className="bg-red-50 border-t border-red-100 px-4 py-2 flex items-center gap-2 shrink-0"
   656→              >
   657→                <AlertTriangle className="size-3.5 text-red-500 shrink-0" />
   658→                <p className="text-xs text-red-600 flex-1">{error}</p>
   659→                <button
   660→                  onClick={() => setError("")}
   661→                  className="text-red-400 hover:text-red-600"
   662→                >
   663→                  <X className="size-3" />
   664→                </button>
   665→              </motion.div>
   666→            )}
   667→          </AnimatePresence>
   668→
   669→          {/* Input Area */}
   670→          {activeChannel && (
   671→            <div className="bg-white border-t border-gray-100 p-3 pb-16 shrink-0">
   672→              {/* Replying to indicator */}
   673→              <AnimatePresence>
   674→                {replyingTo && (
   675→                  <motion.div
   676→                    initial={{ opacity: 0, height: 0 }}
   677→                    animate={{ opacity: 1, height: "auto" }}
   678→                    exit={{ opacity: 0, height: 0 }}
   679→                    className="overflow-hidden"
   680→                  >
   681→                    <div className="flex items-center gap-2 bg-[#F0F4FF] border border-[#1A237E]/10 rounded-lg px-3 py-2 mb-2">
   682→                      <Reply className="size-3 text-[#1A237E] shrink-0" />
   683→                      <div className="min-w-0 flex-1">
   684→                        <p className="text-[10px] font-semibold text-[#1A237E]/60">
   685→                          Replying to {replyingTo.displayName}
   686→                        </p>
   687→                        <p className="text-[10px] text-gray-500 truncate">
   688→                          {replyingTo.content.substring(0, 80)}
   689→                        </p>
   690→                      </div>
   691→                      <button
   692→                        onClick={() => setReplyingTo(null)}
   693→                        className="text-gray-400 hover:text-gray-600 shrink-0"
   694→                      >
   695→                        <X className="size-3.5" />
   696→                      </button>
   697→                    </div>
   698→                  </motion.div>
   699→                )}
   700→              </AnimatePresence>
   701→
   702→              {/* Composer */}
   703→              <div className="flex items-end gap-2">
   704→                <div className="flex-1 relative">
   705→                  <textarea
   706→                    ref={inputRef}
   707→                    value={inputText}
   708→                    onChange={handleInputChange}
   709→                    onKeyDown={handleKeyDown}
   710→                    placeholder={
   711→                      activeChannel.isAnnouncement
   712→                        ? "This channel is for announcements only."
   713→                        : `Message ${activeChannel.name}...`
   714→                    }
   715→                    disabled={activeChannel.isAnnouncement}
   716→                    rows={1}
   717→                    className="w-full resize-none rounded-xl border border-gray-200 bg-[#FAFBFF] px-3.5 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-[#1A237E]/15 focus:border-[#1A237E]/25 transition-all min-h-[42px] max-h-24 disabled:opacity-50 disabled:cursor-not-allowed"
   718→                    style={{ lineHeight: "1.5" }}
   719→                  />
   720→                  {/* Character counter */}
   721→                  {charCount > 0 && (
   722→                    <span
   723→                      className={`absolute bottom-1.5 right-3 text-[9px] ${
   724→                        charCount > MAX_CHARS * 0.9
   725→                          ? "text-red-400"
   726→                          : "text-gray-300"
   727→                      }`}
   728→                    >
   729→                      {charCount}/{MAX_CHARS}
   730→                    </span>
   731→                  )}
   732→                </div>
   733→                <button
   734→                  onClick={sendMessage}
   735→                  disabled={!inputText.trim() || sending || activeChannel.isAnnouncement}
   736→                  className="shrink-0 w-10 h-[42px] rounded-xl bg-[#1A237E] hover:bg-[#283593] text-white flex items-center justify-center transition-colors disabled:opacity-30 disabled:cursor-not-allowed shadow-sm"
   737→                >
   738→                  {sending ? (
   739→                    <Loader2 className="size-4 animate-spin" />
   740→                  ) : (
   741→                    <Send className="size-4" />
   742→                  )}
   743→                </button>
   744→              </div>
   745→            </div>
   746→          )}
   747→        </div>
   748→      </div>
   749→
   750→      {/* ===== REPORT MODAL ===== */}
   751→      <AnimatePresence>
   752→        {reportingMsg && (
   753→          <motion.div
   754→            initial={{ opacity: 0 }}
   755→            animate={{ opacity: 1 }}
   756→            exit={{ opacity: 0 }}
   757→            className="fixed inset-0 bg-black/40 z-50 flex items-center justify-center p-4"
   758→            onClick={() => {
   759→              setReportingMsg(null);
   760→              setReportReason("");
   761→              setReportExplanation("");
   762→            }}
   763→          >
   764→            <motion.div
   765→              initial={{ scale: 0.95, opacity: 0, y: 10 }}
   766→              animate={{ scale: 1, opacity: 1, y: 0 }}
   767→              exit={{ scale: 0.95, opacity: 0, y: 10 }}
   768→              className="bg-white rounded-2xl p-5 max-w-sm w-full shadow-xl"
   769→              onClick={(e) => e.stopPropagation()}
   770→            >
   771→              {/* Header */}
   772→              <div className="flex items-center gap-3 mb-4">
   773→                <div className="w-10 h-10 rounded-full bg-red-50 flex items-center justify-center">
   774→                  <Flag className="size-5 text-red-500" />
   775→                </div>
   776→                <div>
   777→                  <h3 className="text-sm font-bold text-gray-800">
   778→                    Report Message
   779→                  </h3>
   780→                  <p className="text-[10px] text-gray-500">
   781→                    Reviewed confidentially by a church moderator
   782→                  </p>
   783→                </div>
   784→              </div>
   785→
   786→              {/* Reported message preview */}
   787→              <div className="bg-gray-50 rounded-lg p-2.5 mb-4 border border-gray-100">
   788→                <p className="text-[11px] text-gray-600 italic line-clamp-2">
   789→                  &ldquo;{reportingMsg.content.substring(0, 120)}&rdquo;
   790→                </p>
   791→              </div>
   792→
   793→              {/* Reason selection */}
   794→              <div className="mb-4">
   795→                <p className="text-xs font-medium text-gray-700 mb-2">
   796→                  What is the reason?
   797→                </p>
   798→                <div className="space-y-1.5 max-h-48 overflow-y-auto">
   799→                  {REPORT_REASONS.map((r) => (
   800→                    <label
   801→                      key={r.value}
   802→                      className={`flex items-center gap-2.5 px-3 py-2 rounded-lg border cursor-pointer transition-colors ${
   803→                        reportReason === r.value
   804→                          ? "border-[#1A237E] bg-[#1A237E]/5"
   805→                          : "border-gray-100 hover:bg-gray-50"
   806→                      }`}
   807→                    >
   808→                      <div
   809→                        className={`w-4 h-4 rounded-full border-2 flex items-center justify-center transition-colors ${
   810→                          reportReason === r.value
   811→                            ? "border-[#1A237E]"
   812→                            : "border-gray-300"
   813→                        }`}
   814→                      >
   815→                        {reportReason === r.value && (
   816→                          <div className="w-2 h-2 rounded-full bg-[#1A237E]" />
   817→                        )}
   818→                      </div>
   819→                      <span className="text-xs text-gray-700">{r.label}</span>
   820→                    </label>
   821→                  ))}
   822→                </div>
   823→              </div>
   824→
   825→              {/* Optional explanation */}
   826→              {reportReason && (
   827→                <div className="mb-4">
   828→                  <p className="text-xs font-medium text-gray-700 mb-1.5">
   829→                    Additional details (optional)
   830→                  </p>
   831→                  <textarea
   832→                    value={reportExplanation}
   833→                    onChange={(e) => setReportExplanation(e.target.value)}
   834→                    placeholder="Provide any helpful context..."
   835→                    rows={2}
   836→                    className="w-full resize-none rounded-lg border border-gray-200 bg-gray-50 px-3 py-2 text-xs focus:outline-none focus:ring-2 focus:ring-[#1A237E]/15 focus:border-[#1A237E]/25 transition-colors"
   837→                  />
   838→                </div>
   839→              )}
   840→
   841→              {/* Actions */}
   842→              <div className="flex gap-3">
   843→                <button
   844→                  onClick={() => {
   845→                    setReportingMsg(null);
   846→                    setReportReason("");
   847→                    setReportExplanation("");
   848→                  }}
   849→                  className="flex-1 py-2.5 rounded-xl border border-gray-200 text-xs font-medium text-gray-600 hover:bg-gray-50 transition-colors"
   850→                >
   851→                  Cancel
   852→                </button>
   853→                <button
   854→                  onClick={submitReport}
   855→                  disabled={!reportReason || reportSubmitting}
   856→                  className="flex-1 py-2.5 rounded-xl bg-red-500 hover:bg-red-600 text-xs font-medium text-white transition-colors disabled:opacity-40 disabled:cursor-not-allowed flex items-center justify-center gap-1.5"
   857→                >
   858→                  {reportSubmitting ? (
   859→                    <Loader2 className="size-3 animate-spin" />
   860→                  ) : (
   861→                    <Flag className="size-3" />
   862→                  )}
   863→                  Submit Report
   864→                </button>
   865→              </div>
   866→            </motion.div>
   867→          </motion.div>
   868→        )}
   869→      </AnimatePresence>
   870→
   871→      {/* Feature Navigation Bar */}
   872→      <SocialFeatureNav currentFeatureId="family-chat" />
   873→    </div>
   874→  );

   928→                key={emoji}
